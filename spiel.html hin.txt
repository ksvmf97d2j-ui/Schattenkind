                <div class="body girl-body" id="g-body"><div class="arm right girl-arm" id="g-arm-r"></div></div>

        <div id="controls">
            <div class="button-group">
                <button onclick="triggerAction('wave')">Winken 👋</button>
                <button onclick="triggerAction('laugh')">Lachen 😄</button>
                <button onclick="triggerAction('eat')">Eis essen 🍦</button>
                <button onclick="changeOutfit()">Matching Outfit kaufen 👔</button>
            </div>
            <div class="button-group">
                <button onclick="changeLocation()">Ort wechseln 🚶</button>
            </div>
        </div>

    <script>
        let gameHour = 16, gameMinute = 0, delayMs = 1000;
        let currentLocation = "Spielplatz";
        const locations = ["Spielplatz", "Eisdiele", "Bäckerei", "Bekleidungsgeschäft"];
        let locIndex = 0;

        setInterval(() => {
            gameMinute++;
            if (gameMinute >= 60) { gameMinute = 0; gameHour++; if (gameHour >= 24) gameHour = 0; }
            document.getElementById('clock').innerText = (gameHour < 10 ? "0" + gameHour : gameHour) + ":" + (gameMinute < 10 ? "0" + gameMinute : gameMinute);
            const viewport = document.getElementById('viewport');
            if (gameHour >= 20 || gameHour < 6) viewport.classList.add('sky-night');
            else viewport.classList.remove('sky-night');
        }, 1000);

        function showSubtitle(text) {
            const sub = document.getElementById('subtitles');
            sub.innerText = text;
            setTimeout(() => { if (sub.innerText === text) sub.innerText = ""; }, 2500);
        }

        function triggerAction(action) {
            executePlayerAction(action);
            setTimeout(() => executeGirlAction(action), delayMs);
        }

        function executePlayerAction(action) {
            const pArm = document.getElementById('p-arm-r'), pMouth = document.getElementById('p-mouth');
            if (action === 'wave') { pArm.style.transform = 'rotate(-130deg)'; setTimeout(() => pArm.style.transform = 'rotate(0deg)', 1200); }
            else if (action === 'laugh') { pMouth.style.height = '8px'; setTimeout(() => pMouth.style.height = '2px', 1200); }
            else if (action === 'eat') showSubtitle("Du isst ein Bällchen Eis...");
        }

        function executeGirlAction(action) {
            const gArm = document.getElementById('g-arm-r'), gMouth = document.getElementById('g-mouth');
            const isNight = (gameHour >= 20 || gameHour < 6);

            if (action === 'wave') { gArm.style.transform = 'rotate(-130deg)'; setTimeout(() => gArm.style.transform = 'rotate(0deg)', 1200); }
            else if (action === 'laugh') {
                if (isNight) {
                    gMouth.classList.add('creepy-mouth');
                    showSubtitle("Das Mädchen lacht... aber ihr Mund öffnet sich viel zu weit!", 3000);
                    setTimeout(() => gMouth.classList.remove('creepy-mouth'), 2500);
                } else {
                    gMouth.style.height = '8px'; setTimeout(() => gMouth.style.height = '2px', 1200);
                }
            } else if (action === 'eat') {
                if (isNight) showSubtitle("Das Mädchen verschluckt das Eis im Ganzen...", 3000);
                else showSubtitle("Das Mädchen isst ihr Eis (1 Sekunde später).");
            }
        }

        function changeLocation() {
            locIndex = (locIndex + 1) % locations.length;
            currentLocation = locations[locIndex];
            document.getElementById('location').innerText = currentLocation;
            showSubtitle("Ort gewechselt: " + currentLocation);
            const ground = document.getElementById('ground');
            if (currentLocation === "Spielplatz") ground.style.background = "#111a10";
            else if (currentLocation === "Eisdiele") ground.style.background = "#3a2518";
            else if (currentLocation === "Bäckerei") ground.style.background = "#4a3c28";
            else if (currentLocation === "Bekleidungsgeschäft") ground.style.background = "#222530";
        }

        function changeOutfit() {
            if (currentLocation !== "Bekleidungsgeschäft") {
                showSubtitle("Gehe zuerst zum Bekleidungsgeschäft!");
                return;
            }
            document.getElementById('g-body').style.background = "#34495e";
            document.getElementById('g-arm-r').style.background = "#2c3e50";
            showSubtitle("Matching Outfit gekauft! Ihr tragt dieselbe Kleidung.");
        }
    </script>
